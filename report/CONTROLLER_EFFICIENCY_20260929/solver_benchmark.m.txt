entries=struct([]);
for name=["braking346","oncoming8","oncoming15"]
 p=load(fullfile(out,"cone-"+name+".mat"));
 for repetition=1:2
  for method=["auto","normal","schur","augmented"]
   options=p.options;options.LinearSolver=method;options.MaxTime=3;
   wall=tic;[x,value,flag,output]=coneprog(p.objective,p.cones,p.a,p.b,p.equal,p.rhs,p.lower,p.upper,options);seconds=toc(wall);
   linear=Inf;cone=Inf;
   if ~isempty(x)
    linear=max([0;p.a*x-p.b;abs(p.equal*x-p.rhs);p.lower-x;x-p.upper]);cone=0;
    for j=1:numel(p.cones),c=p.cones(j);cone=max(cone,norm(c.A*x-c.b)-c.d.'*x+c.gamma);end
   end
   entry=struct('problem',name,'repetition',repetition,'method',method,'seconds',seconds,'objective',value, ...
    'exitFlag',flag,'output',output,'linearResidual',linear,'coneResidual',cone);
   if isempty(entries),entries=entry;else,entries(end+1)=entry;end
   fprintf('%s %s: %.4f s, flag %d, linear %.3g cone %.3g\n',name,method,seconds,flag,linear,cone);
   f=fopen(fullfile(out,'linear-solver-benchmark.json'),'w');fprintf(f,'%s\n',jsonencode(entries));fclose(f);
  end
 end

function ego=localEgo(x,time,input)
    ego=struct('position',x(1:2),'yaw',x(3),'speed',x(4),'lateralVelocity',x(5),'yawRate',x(6), ...
        'stateTime',time,'heldActuatorInput',input);
end
function target=localTarget(q)
    direction=[cos(q(3)+q(6));sin(q(3)+q(6))];velocity=q(4)*direction;
    yawRate=q(4)*sin(q(6))/q(7);
    target=struct('targetPositionInertial',q(1:2),'targetVelocityInertial',velocity, ...
        'targetYawInertial',q(3),'targetYawRate',yawRate,'targetSideslip',q(6), ...
        'targetTangentialAcceleration',q(5),'targetRearAxleDistance',q(7), ...
        'targetAccelerationInertial',q(5)*direction+yawRate*[-velocity(2);velocity(1)], ...
        'targetLength',2*q(8),'targetWidth',2*q(9),'targetRectangleOffset',q(10:11));
end
