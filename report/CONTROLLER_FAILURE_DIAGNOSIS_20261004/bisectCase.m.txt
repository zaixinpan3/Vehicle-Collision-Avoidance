function bisectCase(root,tag)
    cd(root);addpath('controller','config','scripts');
    for name=["turningCrossing","curvedCrossing"]
        cfg=struct('referenceSpeed',8,'controller',struct('horizonSteps',8));
        r=runNonlinearPredictiveSafetyValidation(Scenarios=name,Frames=1500,ControllerConfiguration=cfg, ...
            RequireCollisionThreat=true,RecoveryDwellSeconds=1).results;
        fprintf('BISECT %s %s holds=%d gap=%.4f recovered=%d failure=%s\n',tag,name,r.executedFrames, ...
            r.minimumReplayClearanceMeters,r.recovery.recovered,r.failure);
    end
end
