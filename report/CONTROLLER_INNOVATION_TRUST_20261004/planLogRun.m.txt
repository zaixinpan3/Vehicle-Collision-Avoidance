global PLAN_LOG; PLAN_LOG={};
cd(fullfile(fileparts(mfilename('fullpath')),'wt-split'));addpath('controller','config','scripts');
c=estimatorControllerIntegrationConfig(); c.randomSeed=20261003;
cfg=struct('referenceSpeed',8,'controller',struct('horizonSteps',8));
r=runNonlinearPredictiveSafetyValidation(Scenarios="turningCrossing",Frames=10,ControllerConfiguration=cfg, ...
    RequireCollisionThreat=false,RecoveryDwellSeconds=1,EstimatorConfiguration=c).results;
fprintf('frames=%d failure=%s\n',r.executedFrames,r.failure);
cfgFull=collisionAvoidanceControllerConfig(cfg);
tire=modifiedFialaTire.parameters(cfgFull); sat=atan(3*tire.longitudinalForceScale./tire.corneringStiffness);
for k=1:numel(PLAN_LOG)
  L=PLAN_LOG{k}; X=L.state.stateTrajectory; U=L.state.inputTrajectory;
  slipF=atan2(X(5,1:end-1)+cfgFull.vehicle.lf*X(6,1:end-1),X(4,1:end-1))-U(1,:);
  slipR=atan2(X(5,1:end-1)-cfgFull.vehicle.lr*X(6,1:end-1),X(4,1:end-1));
  % replay of this plan from its own start (nonlinear) vs affine prediction
  x=X(:,1);err=zeros(1,size(U,2));
  for i=1:size(U,2), x=nonlinearBicycleModel.sample(x,U(:,i),cfgFull); d=x-X(:,i+1); err(i)=norm(d(1:2))+2.6*abs(d(3)); end
  [emax,imax]=max(err);
  fprintf('t=%.2f trust=%.3f corr=%.3f | max|slipF|/sat=%.2f max|slipR|/sat=%.2f | max|brake|=%.3f | replay err max %.3f m at node %d (t+%.2fs) | steer range [%.3f %.3f]\n', ...
    L.time,L.state.trust.scale,L.state.trust.correction,max(abs(slipF))/sat(1),max(abs(slipR))/sat(2),max(abs(U(2,:))),emax,imax,imax*0.05,min(U(1,:)),max(U(1,:)));
end
