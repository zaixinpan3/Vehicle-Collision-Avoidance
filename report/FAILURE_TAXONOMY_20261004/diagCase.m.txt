function diagCase(file,label)
repo='/home/zai/Downloads/ResearchProjects/collisionAvoidance';
addpath(fullfile(repo,'config'),fullfile(repo,'scripts'),fullfile(repo,'estimator'),fullfile(repo,'solver','controller'));
addpath(fullfile(fileparts(mfilename('fullpath')),'controller'),'-begin');
global DIAG_DROP DIAG_SEARCH DIAG_H
s=load(file).snapshot; out=sprintf('CASE %s t=%.2f frame=%d',label,s.time,s.frame);
for drop={"","path","collision","terminalTarget","terminalRoad","stateLimits"}
  DIAG_DROP=drop{1};DIAG_H=zeros(0,3);collisionAvoidanceController("resetNominalTrajectory");
  try
    [~,~,p]=collisionAvoidanceController(s.egoInput,s.targetEstimate,s.road,s.configuration,s.previousState);
    r=sprintf('OK(N=%d,slack=%.2g)',p.metadata.horizonSteps,p.metadata.predictiveBarrierValue);
  catch
    a=DIAG_SEARCH.attempts; st=DIAG_SEARCH.stages; code='';
    if ~isempty(st), code=sprintf('st%d',st(end).solverInfo.nativeStatus); end
    r=sprintf('FAIL(%s %s; inits=%s)',DIAG_SEARCH.terminationReason,code,strjoin(extractBefore([a.initialization]+"   ",6),'/'));
  end
  if drop{1}=="", h=DIAG_H; hs=strjoin(compose('N=%d dist=%.1f lat=%.2f',h(:,1),h(:,2),h(:,3)),' | '); out=sprintf('%s\n   anchors: %s\n   none: %s',out,hs,r);
  else, out=sprintf('%s\n   drop %-14s %s',out,drop{1},r); end
end
fprintf('%s\n',out);
end
