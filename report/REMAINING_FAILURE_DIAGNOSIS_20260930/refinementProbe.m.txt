it=c.diagnostic.iterates{19};before=it.baseline;model=it.model;model.collisionRefinement=model.collisionRefinement | it.candidate.interiorViolation>0;
[after,~]=evaluateFailureInputs(model,before.inputs,before.terminalPhaseMeters);
entry=struct('oldMerit',before.restorationMerit,'oldHard',before.hard,'newMerit',after.restorationMerit,'newHard',after.hard,'newInteriorViolation',max(after.interiorViolation), ...
 'newStageIndices',find(model.collisionRefinement),'candidateMerit',it.candidate.restorationMerit,'radius',it.radius);
f=fopen(fullfile(raw,'refinement-comparison.json'),'w');fprintf(f,'%s\n',jsonencode(entry));fclose(f);fprintf('REFINEMENT old %.6g new %.6g candidate %.6g\n',entry.oldMerit,entry.newMerit,entry.candidateMerit);

f=fopen(fullfile(raw,'bounded-minimum-norm.json'),'w');fprintf(f,'%s\n',jsonencode(entry));fclose(f);fprintf('MINNORM %.6g -> %.6g\n',best.hard,solution.hard);
model=c.diagnostic.model;model.cfg.solver.timeLimitSeconds=30;

f=fopen(fullfile(raw,'rebased-restart.json'),'w');fprintf(f,'%s\n',jsonencode(entry));fclose(f);fprintf('REBASED accepted %d %.3f seconds %d solves\n',entry.accepted,seconds,search.solverCalls);

 for budget=budgets

  save(fullfile(raw,"unaccepted-"+name+"-budget"+budget+".mat"),'solution','model');
 end
