cmake_minimum_required(VERSION 3.18)
project(collision_avoidance LANGUAGES C CXX)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-ffp-contract=off)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
foreach(part target geometry)
    file(GLOB sources CONFIGURE_DEPENDS "${GENERATED_DIR}/${part}/*.c")
    add_library(pcbf_${part} SHARED ${sources})
    target_include_directories(pcbf_${part} PUBLIC "${GENERATED_DIR}/${part}")
    target_link_options(pcbf_${part} PRIVATE -Wl,-Bsymbolic)
endforeach()
target_sources(pcbf_target PRIVATE nrmm_track.c)
add_library(pcbf_flow SHARED "${GENERATED_DIR}/generated/pcbf_kernels.c")
target_include_directories(pcbf_flow PUBLIC "${GENERATED_DIR}/generated")
add_library(pcbf_solver SHARED pcbf_ipopt.cpp)
target_include_directories(pcbf_solver PRIVATE "${CASADI_ROOT}/include")
find_library(CASADI_LIBRARY casadi PATHS "${CASADI_ROOT}" NO_DEFAULT_PATH REQUIRED)
target_link_libraries(pcbf_solver PRIVATE "${CASADI_LIBRARY}")
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_INCLUDES "${CASADI_ROOT}/include")
set(CMAKE_REQUIRED_LIBRARIES "${CASADI_LIBRARY}")
check_cxx_source_compiles("#include <casadi/casadi.hpp>\nint main(){return casadi::has_nlpsol(\"ipopt\");}" CASADI_CURRENT_ABI)
if(NOT CASADI_CURRENT_ABI)
    target_compile_definitions(pcbf_solver PRIVATE _GLIBCXX_USE_CXX11_ABI=0)
endif()
find_path(EIGEN_INCLUDE Eigen/Core HINTS "${CASADI_ROOT}/include/eigen3" PATH_SUFFIXES eigen3 REQUIRED)
add_executable(collision-avoidance app/main.cpp app/model.cpp app/planner.cpp)
target_include_directories(collision-avoidance PRIVATE . "${EIGEN_INCLUDE}")
target_link_libraries(collision-avoidance PRIVATE pcbf_geometry pcbf_flow pcbf_solver pcbf_target)
install(TARGETS collision-avoidance pcbf_geometry pcbf_flow pcbf_solver pcbf_target)
install(FILES "${GENERATED_DIR}/generated/controller.json" "${GENERATED_DIR}/generated/configuration.mat" DESTINATION share)
file(READ "${GENERATED_DIR}/generated/controller.json" PCBF_PROFILE)
configure_file(app/profile.hpp.in profile.hpp @ONLY)
target_include_directories(collision-avoidance PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
